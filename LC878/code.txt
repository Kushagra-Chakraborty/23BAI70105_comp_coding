class Solution:
    def nthMagicalNumber(self, n: int, a: int, b: int) -> int:
        left, right = 1, n * min(a, b)
        lcm_ab = math.lcm(a,b)
        mod = 10**9 +7
        while left < right:
            mid = (right + left) // 2
            count = mid//a + mid//b - mid//lcm_ab
            if count<n:
                left=mid+1
            else:
                right=mid
        return left % mod
