#include <iostream>
#include <string>
#include <vector>
 
using namespace std;
 
// Maximum possible nodes equals the maximum total length of all strings + 1 (for root)
const int MAX_NODES = 600005;
 
// Array-based Trie for O(1) cache-friendly lookups
int trie[MAX_NODES][3];
bool is_end[MAX_NODES];
int node_count = 1; // 0 is reserved for the root
 
// Standard Trie Insertion
void insert(const string& s) {
    int u = 0;
    for (char c : s) {
        int idx = c - 'a';
        if (!trie[u][idx]) {
            trie[u][idx] = node_count++;
        }
        u = trie[u][idx];
    }
    is_end[u] = true;
}
 
// DFS with a modification constraint
bool search(int u, const string& s, int pos, bool changed) {
    // Base case: Reached the end of the query string
    if (pos == s.length()) {
        return changed && is_end[u];
    }
 
    int current_char_idx = s[pos] - 'a';
    
    // Path 1: Continue exactly matching the string
    if (trie[u][current_char_idx]) {
        if (search(trie[u][current_char_idx], s, pos + 1, changed)) {
            return true;
        }
    }
    
    // Path 2: If we haven't used our 1 allowed modification, try branching
    if (!changed) {
        for (int i = 0; i < 3; ++i) {
            // Only try characters that are different from the current string character
            if (i != current_char_idx && trie[u][i]) {
                if (search(trie[u][i], s, pos + 1, true)) {
                    return true;
                }
            }
        }
    }
    
    return false;
}
 
int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
 
    int n, m;
    if (!(cin >> n >> m)) return 0;
 
    string s;
    for (int i = 0; i < n; ++i) {
        cin >> s;
        insert(s);
    }
 
    for (int i = 0; i < m; ++i) {
        cin >> s;
        if (search(0, s, 0, false)) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
 
    return 0;
}
